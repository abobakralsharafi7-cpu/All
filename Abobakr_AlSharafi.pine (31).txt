// منطق CISD — بمرحلتين بنافذة (على فريم التأكيد):
// شراء: نافذة من 5 شموع تبدأ من أدنى شمعة ملامسة رجوعاً: نبحث عن الشموع الحمراء داخلها ونعتمد
//        الحمراء الأدنى قاعاً (وليس أول حمراء). إن لم توجد حمراء في النافذة نوسع البحث رجوعاً
//        نافذة إضافية (5 شموع كل مرة، حتى 50 شمعة). بعد اعتماد المرجعية نرجع للخلف قبلها
//        عن أول شمعة خضراء (شمعة CISD) والمستوى = قمة جسمها. التأكيد: إغلاق شمعة فوق المستوى.
// بيع: نفس المنطق معكوساً: نافذة 5 شموع من أعلى شمعة ملامسة رجوعاً، نعتمد الخضراء الأعلى قمة،
//        ثم قبلها أول شمعة حمراء والمستوى = قاع جسمها. التأكيد: إغلاق شمعة تحت المستوى.
// وإذا حدثت ملامسة أدنى (شراء) أو أعلى (بيع): يُحدَّث المرجع ويبدأ الفحص من جديد.
f_get_dynamic_cisd(bool is_buy, int sweep_idx) =>
    float level = na
    if not na(sweep_idx)
        int dist = bar_index - sweep_idx
        if dist >= 0 and dist < 100
            // المرحلة 1: نافذة 5 شموع — الحمراء الأدنى قاعاً (شراء) / الخضراء الأعلى قمة (بيع) مع التوسع رجوعاً
            int ref_idx = na
            float ref_val = na
            int win_start = dist
            while na(ref_idx) and win_start <= dist + 50
                int win_end = math.min(win_start + 4, dist + 50)
                for i = win_start to win_end
                    if is_buy
                        if close[i] < open[i]
                            if na(ref_val) or low[i] < ref_val
                                ref_val := low[i]
                                ref_idx := i
                    else
                        if close[i] > open[i]
                            if na(ref_val) or high[i] > ref_val
                                ref_val := high[i]
                                ref_idx := i
                win_start += 5
            // المرحلة 2: الرجوع للخلف قبل المرجعية للبحث عن شمعة CISD — خضراء (شراء) / حمراء (بيع)
            if not na(ref_idx)
                for i = ref_idx + 1 to ref_idx + 50
                    if is_buy
                        if close[i] > open[i]
                            level := math.max(open[i], close[i])
                            break
                    else
                        if close[i] < open[i]
                            level := math.min(open[i], close[i])
                            break
    level

f_bos_engine(int pivot_len) =>
    float ph = ta.pivothigh(high, pivot_len, pivot_len)
    float pl = ta.pivotlow(low, pivot_len, pivot_len)
    var float last_high = na
    var float last_low = na
    var bool down_active = false
    var bool up_active = false
    var float liq_sell = na
    var float liq_buy = na
    var bool sweep_sell = false
    var bool sweep_buy = false
    if not na(ph)
        last_high := ph
    if not na(pl)
        last_low := pl
    // BOS هابط: إغلاق تحت آخر قاع محوري (مرة واحدة لكل اختراق) -> تسجيل قمة السيولة
    if not na(last_low) and close < last_low and not down_active
        down_active := true
        liq_sell := last_high
        sweep_sell := false
    if not na(last_low) and close > last_low
        down_active := false
    // سحب السيولة للبيع: العودة واختراق قمة السيولة
    if not na(liq_sell) and high >= liq_sell and not sweep_sell
        sweep_sell := true
    // BOS صاعد: إغلاق فوق آخر قمة محورية (مرة واحدة لكل اختراق) -> تسجيل قاع السيولة
    if not na(last_high) and close > last_high and not up_active
        up_active := true
        liq_buy := last_low
        sweep_buy := false
    if not na(last_high) and close < last_high
        up_active := false
    // سحب السيولة للشراء: العودة وكسر قاع السيولة
    if not na(liq_buy) and low <= liq_buy and not sweep_buy
        sweep_buy := true
    [liq_sell, sweep_sell, liq_buy, sweep_buy]
